% Explicit symbol definition marker (metadata/symbol-census/ equation-chain graphs).
% \symboldef{math} or \symboldef[canonical-id]{math} — canonical-id matches metadata/notation.md.
\makeatletter
\def\symboldef{%
  \@ifnextchar[{\symboldef@opt}{\symboldef@plain}%
}
\def\symboldef@opt[#1]#2{\ensuremath{#2}}
\def\symboldef@plain#1{\ensuremath{#1}}

}

% Bibliography entries start with a one-line reader-facing gloss, keyed by
% BibLaTeX entry key so biber does not need to parse custom data fields.
\input{references/bibliography-summaries.tex}

  \ifcsname bibsummary@\thefield{entrykey}\endcsname
    \csname bibsummary@\thefield{entrykey}\endcsname\par\nobreak\smallskip
  \fi}

% Always append DOI, eprint, and URL at the end of each bibliography entry.
\renewbibmacro*{doi+eprint+url}{}
\renewbibmacro*{finentry}{%
  \iffieldundef{doi}{}{%
    \newunit\newblock
    \printfield{doi}}%
  \iffieldundef{eprinttype}{}{%
    \newunit\newblock
    \usebibmacro{eprint}}%
  \iffieldundef{url}{}{%
    \newunit\newblock
    \printfield{url}}%
  \finentry}
